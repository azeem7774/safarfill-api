// SafarFill admin dashboard (runs in the browser, served from /admin/app.js).
(() => {
  const STORE_LINK = 'https://chromewebstore.google.com/detail/jmmmjippjhpeepdpbkeejcomcidlffol';
  const USD_TO_SAR = 3.75;
  const DAY = 86400000;
  const NAMES = { akbar: 'Akbar', alhind: 'Alhind', airsial: 'Air Sial', muqeem: 'Muqeem' };
  const $ = (id) => document.getElementById(id);
  let adminKey = '';
  try { adminKey = sessionStorage.getItem('sfAdmin') || ''; } catch (e) { /* storage blocked */ }
  let data = null;
  let current = null;

  const esc = (s) => String(s == null ? '' : s).replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
  const fmtDate = (t) => (t ? new Date(t).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' }) : '');
  const fmtTime = (t) => (t ? new Date(t).toLocaleString('en-GB', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' }) : '');
  const ago = (t) => {
    if (!t) return '<span class="muted">never</span>';
    const m = Math.round((Date.now() - t) / 60000);
    if (m < 2) return 'just now';
    if (m < 60) return `${m} min ago`;
    if (m < 1440) return `${Math.round(m / 60)} h ago`;
    return `${Math.round(m / 1440)} days ago`;
  };

  function toast(text) {
    const m = $('toast');
    m.textContent = text;
    m.hidden = false;
    clearTimeout(toast.t);
    toast.t = setTimeout(() => { m.hidden = true; }, 3800);
  }

  async function api(path, body) {
    const res = await fetch(path, {
      method: body ? 'POST' : 'GET',
      headers: { authorization: `Bearer ${adminKey}`, ...(body ? { 'content-type': 'application/json' } : {}) },
      body: body ? JSON.stringify(body) : undefined,
    });
    const j = await res.json().catch(() => ({ ok: false, message: `Server error ${res.status}` }));
    if (res.status === 401) { signOut(); throw new Error(j.message || 'Wrong admin password.'); }
    if (!j.ok) throw new Error(j.message || 'Failed.');
    return j;
  }

  function signOut() {
    adminKey = '';
    try { sessionStorage.removeItem('sfAdmin'); } catch (e) { /* ignore */ }
    $('app').hidden = true; $('login').hidden = false; $('logout').hidden = true; $('refreshTop').hidden = true; $('who').textContent = '';
  }

  // ---------- licence helpers
  const expired = (l) => l.expires_at && l.expires_at <= data.now;
  const planPrice = (plan) => (data.prices && data.prices[plan]) || 0;
  const price = (l) => (l.price != null ? l.price : planPrice(l.plan));
  // Same sum as quote() on the server: plan + add-ons − discount.
  function quoteOf(lic, discount) {
    const r = data.rates; const d = data.plans[lic.plan]; const base = planPrice(lic.plan);
    if (!d || !base) return { lines: [], total: 0 };
    const lines = [[`${lic.plan[0].toUpperCase()}${lic.plan.slice(1)} plan`, base]];
    const portals = lic.plan === 'starter' ? Math.max(0, lic.portals.length - 1) : 0;
    if (portals) lines.push([`+ ${portals} extra portal${portals > 1 ? 's' : ''}`, portals * r.portal]);
    const pcs = Math.max(0, (+lic.max_devices || 1) - d.max_devices);
    if (pcs) lines.push([`+ ${pcs} extra computer${pcs > 1 ? 's' : ''}`, pcs * r.pc]);
    const ai = +lic.ai_quota;
    const packs = d.ai_quota > 0 && ai > d.ai_quota ? Math.ceil((ai - d.ai_quota) / 100) : 0;
    if (packs) lines.push([`+ ${packs * 100} AI reads`, packs * r.ai100]);
    const sub = lines.reduce((a, l) => a + l[1], 0);
    const m = String(discount || '').replace(/\s/g, '').match(/^(\d+(?:\.\d+)?)(%?)$/);
    const off = Math.min(sub, m ? (m[2] ? Math.round(sub * Math.min(100, +m[1]) / 100) : Math.round(+m[1])) : 0);
    if (off) lines.push([`− Discount${m[2] ? ` ${m[1]}%` : ''}`, -off]);
    return { lines, total: sub - off };
  }
  function drawQuote(id, lic, discount) {
    const q = quoteOf(lic, discount);
    $(id).innerHTML = q.lines.length
      ? q.lines.map((l) => `<div class="ql"><span>${esc(l[0])}</span><span>${l[1]}</span></div>`).join('') + `<div class="ql qt"><span>Monthly price</span><span>${q.total} SAR</span></div>`
      : '<div class="ql"><span class="muted">Free (no monthly price)</span></div>';
  }
  const createQuote = () => drawQuote('cQuote', { plan: $('cPlan').value, portals: portalsOf('cPortals'), max_devices: $('cPcs').value, ai_quota: $('cAi').value }, $('cDisc').value);
  const editQuote = () => current && drawQuote('eQuote', { plan: current.plan, portals: portalsOf('ePortals'), max_devices: $('eDevices').value, ai_quota: $('eAi').value }, $('eDisc').value);
  const live = (l) => l.status === 'active' && !expired(l);

  function statusPill(l) {
    if (l.status === 'blocked') return '<span class="pill p-bad">Blocked</span>';
    if (l.plan === 'trial' && !l.activated_at) return '<span class="pill p-muted">Not started</span>';
    if (expired(l)) return `<span class="pill p-bad">${l.plan === 'trial' ? 'Trial over' : 'Expired'}</span>`;
    if (l.plan === 'trial') return '<span class="pill p-ok">On trial</span>';
    if (l.expires_at && l.expires_at - data.now < 5 * DAY) return '<span class="pill p-warn">Ending soon</span>';
    return '<span class="pill p-ok">Active</span>';
  }
  function expiryCell(l) {
    if (l.plan === 'owner' || (!l.expires_at && l.plan !== 'trial')) return '<span class="muted">never</span>';
    if (!l.expires_at) return `<span class="muted">${l.trial_days} days from first use</span>`;
    const left = Math.ceil((l.expires_at - data.now) / DAY);
    return `${fmtDate(l.expires_at)} <span class="muted small">(${left > 0 ? `${left} d left` : `ended ${Math.max(1, -left)} d ago`})</span>`;
  }
  function aiCell(l) {
    const used = l.ai_count || 0;
    if (l.ai_quota < 0) return `${used} <span class="muted small">no limit</span>`;
    if (l.ai_quota === 0) return '<span class="muted small">off · built-in only</span>';
    const p = Math.min(100, Math.round((used / l.ai_quota) * 100));
    return `${used} / ${l.ai_quota}<span class="bar"><i class="${p >= 100 ? 'full' : p >= 85 ? 'hi' : ''}" style="width:${p}%"></i></span>`;
  }

  // ---------- needs your attention
  function attentionItems() {
    const items = [];
    for (const e of data.open) {
      const who = e.agency || 'Unknown agency';
      if (e.code === 'ai_unavailable') {
        items.push({ lvl: 'bad', title: 'AI reader is failing for everyone', text: 'Anthropic refused the key or the credit is empty. Top up credit in the Anthropic console, or check ANTHROPIC_KEY.', acts: [['Mark fixed', () => dismiss(e.id)]] });
      } else if (e.code === 'device_limit') {
        items.push({ lvl: 'info', title: `${who} wants another computer`, text: 'A new computer, or Chrome was reinstalled. Allow one more (+25 SAR/month), or remove the old computer.', acts: [['+1 computer', () => act(e.key, 'add_device_slot')], ['Open', () => openDetail(e.key)], ['Dismiss', () => dismiss(e.id)]] });
      } else if (e.code === 'ai_not_in_trial') {
        items.push({ lvl: 'info', title: `${who} wants to try AI reading`, text: 'Give a small AI test if they insist.', acts: [['Give 5', () => act(e.key, 'ai_test', { amount: 5 })], ['Give 10', () => act(e.key, 'ai_test', { amount: 10 })], ['Dismiss', () => dismiss(e.id)]] });
      } else if (e.code === 'portal_not_in_plan') {
        items.push({ lvl: 'info', title: `${who} tried a portal not in their plan`, text: e.what, acts: [['Open', () => openDetail(e.key)], ['Dismiss', () => dismiss(e.id)]] });
      } else if (e.code === 'device_removed') {
        items.push({ lvl: 'warn', title: `${who}: a removed computer is still trying`, text: 'If this is wrong, open the licence and allow the computer again.', acts: [['Open', () => openDetail(e.key)], ['Dismiss', () => dismiss(e.id)]] });
      }
    }
    for (const l of data.licences) {
      if (l.status === 'blocked') continue;
      const used = l.ai_count || 0;
      if (price(l) && expired(l) && data.now - l.expires_at < 30 * DAY) {
        items.push({ lvl: 'bad', title: `${l.agency || l.key}: licence expired`, text: `Ended ${fmtDate(l.expires_at)}; SafarFill has stopped for them. Collect ${price(l)} SAR, then add 30 days.`, acts: [['Payment received', () => act(l.key, 'extend', { days: 30 })], ['Open', () => openDetail(l.key)]] });
      } else if (l.plan === 'trial' && expired(l) && data.now - l.expires_at < 7 * DAY) {
        items.push({ lvl: 'warn', title: `${l.agency || l.key}: trial over`, text: `Read ${l.parse_count || 0} times during the trial. A good moment to send prices.`, acts: [['Open', () => openDetail(l.key)]] });
      } else if (price(l) && l.expires_at && !expired(l) && l.expires_at - data.now < 5 * DAY) {
        items.push({ lvl: 'warn', title: `${l.agency || l.key}: ends in ${Math.ceil((l.expires_at - data.now) / DAY)} days`, text: `Remind them to pay ${price(l)} SAR before ${fmtDate(l.expires_at)}.`, acts: [['Payment received', () => act(l.key, 'extend', { days: 30 })]] });
      }
      if (live(l) && l.ai_quota > 0 && used >= l.ai_quota) {
        items.push({ lvl: 'bad', title: `${l.agency || l.key}: AI reads used up`, text: `All ${l.ai_quota} used. The built-in reader still works.`, acts: [[l.plan === 'trial' ? 'Give 10' : '+100 AI reads', () => act(l.key, l.plan === 'trial' ? 'ai_test' : 'add_ai', { amount: l.plan === 'trial' ? 10 : 100 })]] });
      } else if (live(l) && l.ai_quota > 0 && l.plan !== 'trial' && used / l.ai_quota >= 0.85) {
        items.push({ lvl: 'warn', title: `${l.agency || l.key}: ${used} of ${l.ai_quota} AI reads used`, text: 'Close to the monthly limit.', acts: [['+100 AI reads', () => act(l.key, 'add_ai', { amount: 100 })]] });
      }
    }
    const order = { bad: 0, warn: 1, info: 2 };
    return items.sort((a, b) => order[a.lvl] - order[b.lvl]);
  }

  function drawAttention() {
    const box = $('attn');
    const items = attentionItems();
    box.innerHTML = items.length ? '' : '<div class="muted">Nothing needs you right now.</div>';
    for (const it of items) {
      const div = document.createElement('div');
      div.className = 'attn-item';
      div.innerHTML = `<span class="stripe s-${it.lvl}"></span><div><b>${esc(it.title)}</b><span class="small muted">${esc(it.text)}</span></div>`;
      const btns = document.createElement('div');
      btns.className = 'row';
      for (const [label, fn] of it.acts) {
        const b = document.createElement('button');
        b.className = 'small';
        b.textContent = label;
        b.addEventListener('click', () => Promise.resolve(fn()).catch((e) => toast(e.message)));
        btns.appendChild(b);
      }
      div.appendChild(btns);
      box.appendChild(div);
    }
  }

  // ---------- main render
  function render() {
    const q = $('filter').value.trim().toLowerCase();
    const rows = data.licences.filter((l) => !q || [l.key, l.agency, l.contact, l.notes].join(' ').toLowerCase().includes(q));
    $('list').innerHTML = rows.map((l) => `
      <tr class="click${current && current.key === l.key ? ' sel' : ''}" data-key="${esc(l.key)}">
        <td class="mono">${esc(l.key)}</td><td>${esc(l.agency)}</td>
        <td>${esc(l.plan)}</td><td>${price(l) ? `${price(l)} SAR${price(l) !== planPrice(l.plan) ? ` <span class="muted small">${l.discount ? `−${esc(l.discount)}` : 'add-ons'}</span>` : ''}` : '<span class="muted">free</span>'}</td>
        <td>${statusPill(l)}</td><td>${expiryCell(l)}</td><td>${l.devices_used || 0} / ${l.max_devices}</td>
        <td>${aiCell(l)}</td><td>${l.parse_count || 0}</td><td>${l.fill_count || 0}</td><td>${ago(l.last_seen)}</td>
      </tr>`).join('') || '<tr><td colspan="11" class="muted">No licences yet. Create the first key below.</td></tr>';

    const liveOnes = data.licences.filter(live);
    const paid = liveOnes.filter((l) => price(l));
    $('sPaid').textContent = paid.length;
    $('sTrials').textContent = liveOnes.filter((l) => l.plan === 'trial').length;
    $('sIncome').textContent = paid.reduce((s, l) => s + price(l), 0);
    $('sAi').textContent = data.totals.ai;
    $('sFills').textContent = data.totals.fills || 0;
    for (const [id, k] of [['rPortal', 'portal'], ['rPc', 'pc'], ['rAi', 'ai100']]) if (document.activeElement !== $(id)) $(id).value = data.rates[k];
    if (document.activeElement !== $('minVersion')) $('minVersion').value = (data.settings && data.settings.min_version) || '';
    const sup = (data.settings && data.settings.support) || {};
    if (document.activeElement !== $('supWa')) $('supWa').value = sup.whatsapp || '';
    if (document.activeElement !== $('supEmail')) $('supEmail').value = sup.email || '';
    if (document.activeElement !== $('payText')) $('payText').value = (data.settings && data.settings.payment) || '';
    drawQuotes();
    const sar = data.totals.cost * USD_TO_SAR;
    $('sCost').textContent = `${sar.toFixed(sar < 10 ? 2 : 0)} SAR`;
    drawAttention();

    $('feed').innerHTML = data.feed.map((e) => {
      const pill = { bad: '<span class="pill p-bad">refused</span> ', warn: '<span class="pill p-warn">attention</span> ', ok: '<span class="pill p-ok">done</span> ' }[e.kind] || '';
      const who = e.key === 'SYSTEM' ? 'System' : (e.agency || e.key);
      return `<div class="ev"><span class="t">${fmtTime(e.at)} · ${esc(who)}</span><span>${pill}${esc(e.what)}</span>${e.sees ? `<span class="sees"><b>Agent sees:</b> ${esc(e.sees)}</span>` : ''}</div>`;
    }).join('') || '<div class="muted">Nothing has happened yet.</div>';
  }

  async function load() {
    data = await api('/admin/api/licences');
    $('app').hidden = false; $('login').hidden = true; $('logout').hidden = false; $('refreshTop').hidden = false;
    $('who').textContent = `Month ${data.month}`;
    if (!$('cPortals').children.length) planDefaults();
    render();
  }

  // ---------- quotes
  const qNo = (q) => `Q-${String(q.id).padStart(4, '0')}`;
  const qStatus = (q) => (q.status === 'sent' && q.valid_until < data.now ? 'expired' : q.status);
  const portalNames = (ps) => ps.map((p) => NAMES[p] || p).join(', ');
  function drawQuotes() {
    $('quotes').innerHTML = (data.quotes || []).map((q) => {
      const st = qStatus(q);
      const pill = { sent: 'p-warn', accepted: 'p-ok', declined: 'p-muted', expired: 'p-muted' }[st];
      return `<tr><td class="mono">${qNo(q)}</td><td>${fmtDate(q.created_at)}</td><td>${esc(q.agency)}</td><td>${esc(q.plan)}</td><td>${q.price} SAR</td>
        <td>${fmtDate(q.valid_until)}</td><td><span class="pill ${pill}">${st}</span>${q.key ? ` <span class="mono small">${esc(q.key)}</span>` : ''}</td>
        <td><div class="qbtns"><button class="small" data-q="pdf" data-id="${q.id}">PDF</button><button class="small" data-q="copy" data-id="${q.id}">Copy text</button>
        ${q.status !== 'accepted' ? `<button class="small primary" data-q="accept" data-id="${q.id}">Accept → key</button>` : ''}
        ${q.status === 'sent' ? `<button class="small" data-q="decline" data-id="${q.id}">Declined</button>` : ''}</div></td></tr>`;
    }).join('') || '<tr><td colspan="8" class="muted">No quotes yet. Fill in the form above and press "Save as quote".</td></tr>';
  }
  function includes(q) {
    return [
      `Portals: ${portalNames(q.portals)}`,
      `${q.max_devices} computer${q.max_devices > 1 ? 's' : ''}`,
      `${q.ai_quota} AI reads per month`,
      'Built-in passport reader: unlimited',
    ];
  }
  function quoteText(q) {
    const sup = data.settings.support || {};
    const w = Math.max(...q.lines.map((l) => l[0].length)) + 2;
    return [
      `Assalam o Alaikum${q.agency ? ` ${q.agency}` : ''},`, '',
      `SafarFill quote ${qNo(q)}`,
      ...q.lines.map((l) => `${l[0].padEnd(w)}${l[1] < 0 ? '−' + -l[1] : l[1]} SAR`),
      `Total: ${q.price} SAR / month`, '',
      `Includes: ${includes(q).join('; ')}.`,
      `Valid until ${fmtDate(q.valid_until)}.`,
      ...(data.settings.payment ? ['', 'Payment:', data.settings.payment] : []),
      ...(sup.whatsapp ? ['', `WhatsApp: +${sup.whatsapp}`] : []),
    ].join('\n');
  }
  function printQuote(q) {
    const sup = data.settings.support || {};
    const contact = [sup.whatsapp ? `WhatsApp +${esc(sup.whatsapp)}` : '', sup.email ? esc(sup.email) : ''].filter(Boolean).join(' · ');
    $('printArea').innerHTML = `
      <div class="pq-head"><div class="pq-brand">Sahl Labs<small>SafarFill · passport reading and form filling for travel agencies</small></div>
        <div class="pq-title"><b>QUOTATION</b>${qNo(q)}</div></div>
      <div class="pq-meta">
        <div><div class="k">Prepared for</div><b>${esc(q.agency || '—')}</b>${q.contact ? `<div>${esc(q.contact)}</div>` : ''}</div>
        <div style="text-align:right"><div class="k">Date</div>${fmtDate(q.created_at)}<div class="k" style="margin-top:6px">Valid until</div>${fmtDate(q.valid_until)}</div>
      </div>
      <table class="pq-table"><thead><tr><th>Item</th><th class="n">SAR / month</th></tr></thead><tbody>
        ${q.lines.map((l) => `<tr><td>${esc(l[0].replace(/^[+−] /, ''))}</td><td class="n">${l[1] < 0 ? '−' + -l[1] : l[1]}</td></tr>`).join('')}
        <tr class="tot"><td>Total per month</td><td class="n">${q.price} SAR</td></tr></tbody></table>
      <div class="pq-box"><h4>Included</h4><ul>${includes(q).map((i) => `<li>${esc(i)}</li>`).join('')}</ul></div>
      ${data.settings.payment ? `<div class="pq-box"><h4>How to pay</h4><div class="pq-pre">${esc(data.settings.payment)}</div></div>` : ''}
      <div class="pq-foot">Prices are monthly and paid in advance. Your licence key is sent after payment. ${contact ? `Questions: ${contact}` : ''}</div>`;
    const title = document.title;
    document.title = `SafarFill quote ${qNo(q)}${q.agency ? ` - ${q.agency}` : ''}`;
    const back = () => { document.title = title; window.removeEventListener('afterprint', back); };
    window.addEventListener('afterprint', back);
    window.print();
  }
  function showNewKey(key, agency) {
    const msg = message(key);
    $('newKey').innerHTML = `<div class="newkey"><div class="small muted">New key for ${esc(agency || 'this agency')}</div><div class="mono">${esc(key)}</div><div class="wa">${esc(msg)}</div><div class="row" style="margin-top:8px"><button id="copyKey" class="small">Copy key</button><button id="copyMsg" class="small">Copy WhatsApp message</button></div></div>`;
    $('copyKey').addEventListener('click', () => copy(key, 'Key copied.'));
    $('copyMsg').addEventListener('click', () => copy(msg, 'Message copied. Paste it in WhatsApp.'));
  }

  // ---------- detail
  function drawPortals(id, chosen) {
    $(id).innerHTML = data.portals.map((p) => `<label><input type="checkbox" value="${p}" ${chosen.includes(p) ? 'checked' : ''}> ${NAMES[p] || p}</label>`).join('');
  }
  const portalsOf = (id) => [...$(id).querySelectorAll('input:checked')].map((i) => i.value);

  async function openDetail(key) {
    showDetail(await api(`/admin/api/licence?key=${encodeURIComponent(key)}`));
    $('detail').scrollIntoView({ behavior: 'smooth', block: 'start' });
  }

  function showDetail(j) {
    current = j.licence;
    const l = current;
    $('detail').hidden = false;
    $('confirmBox').innerHTML = '';
    $('dTitle').innerHTML = `<span class="mono">${esc(l.key)}</span> · ${esc(l.agency || 'no name')} · ${esc(l.plan)}${l.status === 'blocked' ? ' · <span class="pill p-bad">Blocked</span>' : ''}`;
    $('eAgency').value = l.agency; $('eContact').value = l.contact; $('eDevices').value = l.max_devices; $('eAi').value = l.ai_quota; $('eNotes').value = l.notes; $('eDisc').value = l.discount || '';
    $('paidDisc').value = '';
    drawPortals('ePortals', l.portals);
    editQuote();
    $('aiTest').hidden = l.plan !== 'trial';
    $('addAi').hidden = l.plan === 'trial' || l.ai_quota < 0;
    $('block').hidden = l.status === 'blocked';
    $('unblock').hidden = l.status !== 'blocked';
    $('dDevices').innerHTML = j.devices.map((d) => `<tr><td class="mono">${esc(d.device.slice(0, 8))}…</td><td>${esc(d.version)}</td><td>${ago(d.last_seen)}</td><td><button class="small danger" data-dev="${esc(d.device)}">Remove</button></td></tr>`).join('') || '<tr><td colspan="4" class="muted">Not used on any computer yet.</td></tr>';
    $('dUsage').innerHTML = j.usage.map((u) => `<tr><td>${esc(u.month)}</td><td>${u.parse_count}</td><td>${u.fill_count || 0}</td><td>${u.ai_count}</td><td>${(u.ai_cost_usd * USD_TO_SAR).toFixed(2)} SAR</td></tr>`).join('') || '<tr><td colspan="5" class="muted">No use yet.</td></tr>';
    $('dLog').innerHTML = j.log.map((e) => `<tr><td class="muted small">${fmtTime(e.at)}</td><td class="wrap">${esc(e.what)}</td></tr>`).join('');
    if (data) render();
  }

  async function act(key, action, extra = {}) {
    const j = await api('/admin/api/action', { key, action, ...extra });
    await load();
    if (current && current.key === key) showDetail(j);
    toast('Done.');
  }
  async function dismiss(id) {
    await api('/admin/api/action', { action: 'dismiss', event_id: id });
    await load();
  }

  function askConfirm(text, yes) {
    $('confirmBox').innerHTML = `<div class="confirm"><div>${esc(text)}</div><div class="row" style="margin-top:8px"><button id="cYes" class="small danger">Yes</button><button id="cNo" class="small">Cancel</button></div></div>`;
    $('cYes').addEventListener('click', () => { $('confirmBox').innerHTML = ''; Promise.resolve(yes()).catch((e) => toast(e.message)); });
    $('cNo').addEventListener('click', () => { $('confirmBox').innerHTML = ''; });
  }
  const run = (fn) => () => Promise.resolve(fn()).catch((e) => toast(e.message));

  // ---------- create
  function planDefaults() {
    const plan = $('cPlan').value;
    const p = data.plans[plan];
    $('cDays').value = p.days || '';
    $('cDays').disabled = plan === 'owner';
    $('cPcs').value = p.max_devices;
    $('cAi').value = p.ai_quota;
    drawPortals('cPortals', p.portals);
    createQuote();
  }

    if (!q) return;
    if (b.dataset.q === 'pdf') printQuote(q);
    else if (b.dataset.q === 'copy') copy(quoteText(q), 'Quote copied. Paste it in WhatsApp.');
    else if (b.dataset.q === 'decline') run(async () => { await api('/admin/api/action', { action: 'quote_status', id: q.id, status: 'declined' }); await load(); })();
    else if (b.dataset.q === 'accept') run(async () => {
      const j = await api('/admin/api/action', { action: 'quote_accept', id: q.id });
      await load();
      showNewKey(j.key, q.agency);
      $('newKey').scrollIntoView({ behavior: 'smooth', block: 'center' });
      toast(`Key made from ${qNo(q)} at ${q.price} SAR/month.`);
    })();
